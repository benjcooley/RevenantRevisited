# Widget registry layout (DAT_00655510)
# ============================================================================
# DAT_00655510 : void**          ; array of pointers to registration records
# DAT_0065617c : int             ; count of registered widget classes (= 8)
#
# Each registration record (32-bit fields) :
#   +0x00 : vtable_ptr           ; widget class vtable in .rdata (0x5a3xxx)
#   +0x04 : name_ptr             ; ASCII keyword (BITMAP / SCROLLBAR / ...)
#

# does:
#     DAT_00655510[DAT_0065617c++] = &recordN;
#     *((void**)&recordN) = vtable_ptr;
#     *((const char**)((char*)&recordN + 4)) = "WIDGET_NAME";
#
